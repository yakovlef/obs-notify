cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(compilerconfig)
include(defaults)
include(helpers)

add_library(${CMAKE_PROJECT_NAME} MODULE src/plugin.cpp src/message-format.cpp src/message-format.hpp)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets Qt6::Network)
set_target_properties(${CMAKE_PROJECT_NAME} PROPERTIES AUTOMOC ON AUTOUIC ON AUTORCC ON)
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})

include(CTest)
if(BUILD_TESTING)
  add_executable(${CMAKE_PROJECT_NAME}-tests tests/message-format-test.cpp src/message-format.cpp src/message-format.hpp)
  target_include_directories(${CMAKE_PROJECT_NAME}-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(${CMAKE_PROJECT_NAME}-tests PRIVATE Qt6::Core)
  add_test(NAME message-format COMMAND ${CMAKE_PROJECT_NAME}-tests)
  if(WIN32)
    set_tests_properties(
      message-format PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>"
    )
  endif()
endif()

if(WIN32)
  target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE NOMINMAX)
  target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE Crypt32)
endif()
